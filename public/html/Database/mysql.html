Relational Database Management System

📚 Introduction to MySQL

MySQL is an open-source relational database management system (RDBMS) that uses Structured Query Language (SQL).

Key Concepts

  • Database: Collection of organized data
  • DBMS: Software to manage databases (MySQL, PostgreSQL)
  • SQL: Structured Query Language to interact with databases
  • Database Server: Can contain multiple databases

Database Hierarchy

Application → DBMS → Database → Table → Row → Column

MySQL vs SQL

  • SQL: The language itself (Structured Query Language)
  • MySQL: A database management system that uses SQL

🗄️ Databases & Tables

Database Commands

-- Show all databases SHOW DATABASES; -- Create database CREATE DATABASE company; -- Use database USE company; -- Drop database DROP DATABASE company;

Table Commands

-- Show all tables SHOW TABLES; -- Show columns in table SHOW COLUMNS FROM users; -- Describe table structure DESC users; -- Create table CREATE TABLE users ( id INT, name VARCHAR(255), age INT ); -- Drop table DROP TABLE users;

➕ INSERT - Create Data

Insert Single Row

INSERT INTO users (id, name, age) VALUES (1, 'John', 30);

Insert Multiple Rows

INSERT INTO users (name, age) VALUES ('Alice', 25), ('Bob', 35), ('Charlie', 28);
💡 Tip: If a column has AUTO_INCREMENT, you don't need to specify its value.

🔍 SELECT - Read Data

Basic SELECT

-- Select all columns SELECT * FROM users; -- Select specific columns SELECT name, email FROM users;

WHERE Clause

-- Single condition SELECT * FROM users WHERE id = 1; -- Multiple conditions (AND) SELECT * FROM users WHERE age >= 25 AND age <= 35; -- OR condition SELECT * FROM users WHERE name = 'John' OR name = 'Alice';

ORDER BY & LIMIT

-- Sort ascending SELECT * FROM users ORDER BY age ASC; -- Sort descending SELECT * FROM users ORDER BY age DESC; -- Limit results SELECT * FROM users LIMIT 10; -- Offset and limit SELECT * FROM users LIMIT 10 OFFSET 20;

LIKE & Wildcards

-- Starts with 'J' SELECT * FROM users WHERE name LIKE 'J%'; -- Ends with 'n' SELECT * FROM users WHERE name LIKE '%n'; -- Contains 'oh' SELECT * FROM users WHERE name LIKE '%oh%';

✏️ UPDATE - Modify Data

-- Update single record UPDATE users SET name = 'John Doe', age = 31 WHERE id = 1; -- Update multiple records UPDATE users SET status = 'active' WHERE age >= 18;
⚠️ Warning: Always use WHERE clause with UPDATE, or you'll update ALL rows!

🗑️ DELETE - Remove Data

-- Delete specific record DELETE FROM users WHERE id = 1; -- Delete with condition DELETE FROM users WHERE age < 18; -- Delete all records (DANGER!) DELETE FROM users;
⚠️ Danger: DELETE without WHERE clause removes ALL rows!

CREATE TABLE students ( student_id INT AUTO_INCREMENT PRIMARY KEY, name VARCHAR(100) NOT NULL, email VARCHAR(100) UNIQUE NOT NULL, age INT CHECK (age >= 18), department_id INT, enrollment_date DATE DEFAULT (CURRENT_DATE), -- Foreign key CONSTRAINT fk_department FOREIGN KEY (department_id) REFERENCES departments(department_id) ON DELETE CASCADE, -- Index for faster searches INDEX idx_name (name) );

🔑 Primary & Foreign Keys

Primary Key

Uniquely identifies each record in a table.

CREATE TABLE users ( id INT PRIMARY KEY AUTO_INCREMENT, name VARCHAR(255), age INT ); INSERT INTO users (name, age) VALUES ('John', 30); SELECT * FROM users;

Foreign Key

Links two tables together.

-- Parent table CREATE TABLE users ( id INT PRIMARY KEY AUTO_INCREMENT, name VARCHAR(255) ); -- Child table with foreign key CREATE TABLE posts ( id INT PRIMARY KEY AUTO_INCREMENT, title VARCHAR(255), user_id INT, FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE ); INSERT INTO users (name) VALUES ('John'); INSERT INTO posts (title, user_id) VALUES ('My First Post', 1);
💡 ON DELETE CASCADE: When a user is deleted, all their posts are automatically deleted too.

🔗 Joins

Joins combine rows from two or more tables based on a related column.

INNER JOIN

Returns only matching records from both tables.

SELECT users.name, posts.title FROM users INNER JOIN posts ON users.id = posts.user_id;

LEFT JOIN

Returns all records from left table, and matching records from right table.

SELECT users.name, posts.title FROM users LEFT JOIN posts ON users.id = posts.user_id;

RIGHT JOIN

Returns all records from right table, and matching records from left table.

SELECT users.name, posts.title FROM users RIGHT JOIN posts ON users.id = posts.user_id;

Real-World Example

-- Get users with their orders SELECT users.name, orders.order_id, orders.total_amount, orders.order_date FROM users LEFT JOIN orders ON users.id = orders.user_id WHERE orders.status = 'completed' ORDER BY orders.order_date DESC;

📊 MySQL Functions

Aggregate Functions

-- COUNT - Count rows SELECT COUNT(*) AS total_users FROM users; -- SUM - Total of column SELECT SUM(price) AS total_sales FROM orders; -- AVG - Average SELECT AVG(age) AS avg_age FROM users; -- MIN and MAX SELECT MIN(price), MAX(price) FROM products; -- GROUP BY SELECT city, COUNT(*) AS user_count FROM users GROUP BY city;

String Functions

-- CONCAT - Join strings SELECT CONCAT(first_name, ' ', last_name) AS full_name FROM users; -- UPPER and LOWER SELECT UPPER(name), LOWER(email) FROM users; -- LENGTH SELECT name, LENGTH(name) AS name_length FROM users; -- LEFT and RIGHT SELECT LEFT(name, 3), RIGHT(name, 3) FROM users;

Date/Time Functions

-- NOW - Current date and time SELECT NOW(); -- CURDATE - Current date SELECT CURDATE(); -- CURTIME - Current time SELECT CURTIME(); -- DATE_FORMAT SELECT DATE_FORMAT(created_at, '%Y-%m-%d') FROM users; -- DATEDIFF - Difference between dates SELECT DATEDIFF(NOW(), created_at) AS days_old FROM users; -- DATE_ADD and DATE_SUB SELECT DATE_ADD(NOW(), INTERVAL 7 DAY); SELECT DATE_SUB(NOW(), INTERVAL 1 MONTH);

Other Useful Functions

-- IF statement SELECT name, IF(age >= 18, 'Adult', 'Minor') AS category FROM users; -- CASE statement SELECT name, CASE WHEN age < 18 THEN 'Minor' WHEN age >= 18 AND age < 65 THEN 'Adult' ELSE 'Senior' END AS age_group FROM users; -- ROUND SELECT ROUND(price, 2) FROM products; -- RAND - Random number SELECT RAND(); -- VERSION SELECT VERSION();

⚙️ Operators

Comparison Operators

-- Equal SELECT * FROM users WHERE age = 30; -- Not equal SELECT * FROM users WHERE age != 30; SELECT * FROM users WHERE age <> 30; -- Greater than / Less than SELECT * FROM users WHERE age > 25; SELECT * FROM users WHERE age < 40; -- Greater than or equal / Less than or equal SELECT * FROM users WHERE age >= 18; SELECT * FROM users WHERE age <= 65;

Logical Operators

-- AND SELECT * FROM users WHERE age >= 18 AND age <= 65; -- OR SELECT * FROM users WHERE city = 'New York' OR city = 'Los Angeles'; -- NOT SELECT * FROM users WHERE NOT city = 'Boston'; -- BETWEEN SELECT * FROM users WHERE age BETWEEN 25 AND 35; -- IN SELECT * FROM users WHERE city IN ('New York', 'LA', 'Chicago'); -- IS NULL / IS NOT NULL SELECT * FROM users WHERE email IS NULL; SELECT * FROM users WHERE email IS NOT NULL;

Special Operators

-- DISTINCT - Remove duplicates SELECT DISTINCT city FROM users; -- AS - Alias SELECT name AS user_name, age AS user_age FROM users; -- UNION - Combine results SELECT name FROM users UNION SELECT name FROM customers; -- EXISTS SELECT * FROM users WHERE EXISTS (SELECT * FROM orders WHERE orders.user_id = users.id);
