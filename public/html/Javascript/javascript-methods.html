📐 Math Methods

The Math object provides mathematical constants and functions.

abs() - Absolute Value

Returns the absolute value of a number (always positive)

console.log(Math.abs(7.25)); // 7.25 console.log(Math.abs(-7.25)); // 7.25 console.log(Math.abs(null)); // 0 console.log(Math.abs(2 - 3)); // 1 // Real example: Calculate price difference function priceDifference(price1, price2) { return Math.abs(price1 - price2); } console.log(priceDifference(50, 75)); // 25

ceil() - Round Up

Rounds a number upward to the nearest integer

console.log(Math.ceil(0.60)); // 1 console.log(Math.ceil(0.40)); // 1 console.log(Math.ceil(5.1)); // 6 console.log(Math.ceil(-5.9)); // -5 // Real example: Calculate number of pages function calculatePages(totalItems, itemsPerPage) { return Math.ceil(totalItems / itemsPerPage); } console.log(calculatePages(100, 15)); // 7 pages

floor() - Round Down

Rounds a number downward to the nearest integer

console.log(Math.floor(0.60)); // 0 console.log(Math.floor(5.9)); // 5 console.log(Math.floor(-5.1)); // -6 // Real example: Calculate discount tiers function getDiscountTier(totalSpent) { return Math.floor(totalSpent / 100); // $100 per tier } console.log(getDiscountTier(450)); // Tier 4

round() - Round to Nearest

Rounds to the nearest integer

console.log(Math.round(2.60)); // 3 console.log(Math.round(2.50)); // 3 console.log(Math.round(2.49)); // 2 // Real example: Round rating function displayRating(rating) { return Math.round(rating) + " stars"; } console.log(displayRating(4.7)); // "5 stars"

max() & min() - Find Maximum/Minimum

// max() - Returns highest value console.log(Math.max(5, 10)); // 10 console.log(Math.max(0, 150, 30, 20)); // 150 console.log(Math.max(...[1, 2, 3])); // 3 (spread array) // min() - Returns lowest value console.log(Math.min(5, 10)); // 5 console.log(Math.min(0, 150, 30, 20)); // 0 // Real example: Find cheapest product const prices = [29.99, 19.99, 39.99, 15.99]; const cheapest = Math.min(...prices); console.log(`Lowest price: $${cheapest}`); // $15.99

pow() - Power/Exponentiation

console.log(Math.pow(2, 3)); // 8 (2³) console.log(Math.pow(3, 3)); // 27 (3³) console.log(Math.pow(2, 4)); // 16 (2⁴) // Real example: Calculate compound interest function compoundInterest(principal, rate, years) { return principal * Math.pow(1 + rate, years); } const result = compoundInterest(1000, 0.05, 10); console.log(`Future value: $${result.toFixed(2)}`);

random() - Random Number

// Returns random number between 0 and 1 console.log(Math.random()); // e.g., 0.8764 // Random between 0 and 10 console.log(Math.random() * 10); // Random integer between 1 and 10 console.log(Math.floor(Math.random() * 10) + 1); // Real example: Generate random ID function generateId() { return Math.floor(Math.random() * 1000000); } console.log(`User ID: ${generateId()}`); // Random element from array const colors = ["red", "blue", "green", "yellow"]; const randomColor = colors[Math.floor(Math.random() * colors.length)]; console.log(randomColor);

sqrt() - Square Root

console.log(Math.sqrt(9)); // 3 console.log(Math.sqrt(64)); // 8 console.log(Math.sqrt(-9)); // NaN (not a number) // Real example: Calculate distance function distance(x1, y1, x2, y2) { return Math.sqrt(Math.pow(x2 - x1, 2) + Math.pow(y2 - y1, 2)); } console.log(distance(0, 0, 3, 4)); // 5

trunc() - Remove Decimal

console.log(Math.trunc(4.9)); // 4 console.log(Math.trunc(-4.9)); // -4 console.log(Math.trunc(4.1)); // 4

Real-World Examples

// Validate and format price function formatPrice(value) { if (!Number.isFinite(value)) { return "Invalid price"; } return `$${value.toFixed(2)}`; } console.log(formatPrice(29.99)); // "$29.99" console.log(formatPrice(19.9)); // "$19.90" console.log(formatPrice("abc")); // "Invalid price" // Parse user input function calculateTotal(inputValue) { const num = parseFloat(inputValue); if (Number.isNaN(num)) { return "Please enter a valid number"; } return num * 1.08; // Add 8% tax } console.log(calculateTotal("100")); // 108

📅 Date Methods

The Date object works with dates and times.

Creating Dates

// Current date and time const now = new Date(); console.log(now); // Specific date const birthday = new Date("2000-01-15"); const meeting = new Date(2025, 0, 15, 10, 30); // Jan 15, 2025 10:30

Get Methods

const date = new Date(); console.log(date.getFullYear()); // 2024 console.log(date.getMonth()); // 0-11 (0 = January) console.log(date.getDate()); // 1-31 (day of month) console.log(date.getDay()); // 0-6 (0 = Sunday) console.log(date.getHours()); // 0-23 console.log(date.getMinutes()); // 0-59 console.log(date.getSeconds()); // 0-59 console.log(date.getTime()); // Milliseconds since 1970

Format Methods

const date = new Date(); console.log(date.toString()); // Full date string console.log(date.toDateString()); // "Mon Jan 01 2024" console.log(date.toTimeString()); // "10:30:00 GMT+0000" console.log(date.toISOString()); // "2024-01-01T10:30:00.000Z" console.log(date.toLocaleDateString()); // "1/1/2024" (locale) console.log(date.toLocaleTimeString()); // "10:30:00 AM"

Real-World Examples

// Calculate age function calculateAge(birthDate) { const birth = new Date(birthDate); const today = new Date(); let age = today.getFullYear() - birth.getFullYear(); const monthDiff = today.getMonth() - birth.getMonth(); if (monthDiff < 0 || (monthDiff === 0 && today.getDate() < birth.getDate())) { age--; } return age; } console.log(calculateAge("2000-01-15")); // Current age // Format date for display function formatDate(date) { const months = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"]; const d = new Date(date); return `${months[d.getMonth()]} ${d.getDate()}, ${d.getFullYear()}`; } console.log(formatDate(new Date())); // "Dec 30, 2024" // Days until event function daysUntil(eventDate) { const today = new Date(); const event = new Date(eventDate); const diff = event - today; const days = Math.ceil(diff / (1000 * 60 * 60 * 24)); return days; } console.log(`Days until New Year: ${daysUntil("2025-01-01")}`);

Common Use Cases

  • 📅 Display current date/time: User dashboard, blog posts
  • ⏰ Calculate time differences: Shipping estimates, event countdowns
  • 🎂 Age calculation: User registration, age verification
  • 📊 Date filtering: Reports, analytics, date pickers
  • ⏱️ Timestamps: Activity logs, chat messages, file metadata

Date Set Methods

Modify dates and times:

const date = new Date(); // Set specific parts of date date.setFullYear(2025); date.setMonth(11); // December (0-indexed) date.setDate(25); date.setHours(12); date.setMinutes(30); date.setSeconds(0); console.log(date); // Christmas 2025 at 12:30 PM // Set using timestamp date.setTime(1000000000000); // UTC versions date.setUTCFullYear(2025); date.setUTCMonth(0);

📝 String Methods

The String object provides methods for text manipulation.

Case Conversion

toUpperCase()

Converts string to uppercase

const text = "hello world"; console.log(text.toUpperCase()); // "HELLO WORLD"

toLowerCase()

Converts string to lowercase

const text = "HELLO WORLD"; console.log(text.toLowerCase()); // "hello world"

Search & Extract

indexOf()

Returns position of first occurrence

const text = "Hello World"; console.log(text.indexOf("World")); // 6 console.log(text.indexOf("xyz")); // -1 (not found)

lastIndexOf()

Returns position of last occurrence

const text = "Hello World, World"; console.log(text.lastIndexOf("World")); // 13

includes()

Checks if string contains substring

const text = "JavaScript"; console.log(text.includes("Script")); // true

startsWith() / endsWith()

Checks start or end of string

const url = "https://example.com"; console.log(url.startsWith("https")); // true console.log(url.endsWith(".com")); // true

Extract Substrings

const text = "JavaScript Programming"; // slice(start, end) - extracts part console.log(text.slice(0, 10)); // "JavaScript" console.log(text.slice(-11)); // "Programming" // substring(start, end) - similar to slice console.log(text.substring(0, 10)); // "JavaScript" // substr(start, length) - deprecated console.log(text.substr(0, 10)); // "JavaScript" // charAt(index) - get character at position console.log(text.charAt(0)); // "J" // charCodeAt(index) - get character code console.log(text.charCodeAt(0)); // 74

Modify & Transform

// replace() - replace first occurrence const text = "Hello World, World"; console.log(text.replace("World", "JS")); // "Hello JS, World" // replaceAll() - replace all occurrences console.log(text.replaceAll("World", "JS")); // "Hello JS, JS" // split() - convert string to array const csv = "apple,banana,orange"; console.log(csv.split(",")); // ["apple", "banana", "orange"] // trim() - remove whitespace const messy = " hello "; console.log(messy.trim()); // "hello" console.log(messy.trimStart()); // "hello " console.log(messy.trimEnd()); // " hello" // repeat() - repeat string console.log("*".repeat(5)); // "*****" // padStart() / padEnd() - pad string const num = "5"; console.log(num.padStart(3, "0")); // "005" console.log(num.padEnd(3, "0")); // "500"

Real-World String Examples

// Email validation function isValidEmail(email) { return email.includes("@") && email.includes("."); } // Format phone number function formatPhone(phone) { const cleaned = phone.replace(/\D/g, ""); return `(${cleaned.slice(0,3)}) ${cleaned.slice(3,6)}-${cleaned.slice(6)}`; } console.log(formatPhone("1234567890")); // "(123) 456-7890" // Capitalize words function capitalize(str) { return str.split(" ") .map(word => word.charAt(0).toUpperCase() + word.slice(1)) .join(" "); } console.log(capitalize("hello world")); // "Hello World"

🗂️ Array Methods

Essential methods for working with arrays.

Add/Remove Elements

Method Description Mutates?
push() Add to end ✅ Yes
pop() Remove from end ✅ Yes
unshift() Add to beginning ✅ Yes
shift() Remove from beginning ✅ Yes
splice() Add/remove at any position ✅ Yes
slice() Extract portion ❌ No

Iteration Methods

const numbers = [1, 2, 3, 4, 5]; // forEach() - execute function for each element numbers.forEach(num => console.log(num * 2)); // map() - transform array (returns new array) const doubled = numbers.map(num => num * 2); // [2, 4, 6, 8, 10] // filter() - filter elements (returns new array) const evens = numbers.filter(num => num % 2 === 0); // [2, 4] // find() - find first matching element const found = numbers.find(num => num > 3); // 4 // findIndex() - find index of first match const index = numbers.findIndex(num => num > 3); // 3 // some() - check if any element passes test const hasEven = numbers.some(num => num % 2 === 0); // true // every() - check if all elements pass test const allPositive = numbers.every(num => num > 0); // true // reduce() - reduce array to single value const sum = numbers.reduce((total, num) => total + num, 0); // 15

Other Useful Methods

const arr = [3, 1, 4, 1, 5]; // sort() - sort array (mutates) arr.sort((a, b) => a - b); // [1, 1, 3, 4, 5] // reverse() - reverse array (mutates) arr.reverse(); // [5, 4, 3, 1, 1] // concat() - join arrays const arr2 = [6, 7]; const joined = arr.concat(arr2); // join() - convert to string console.log(arr.join(", ")); // "5, 4, 3, 1, 1" // includes() - check if element exists console.log(arr.includes(3)); // true // indexOf() - find first index console.log(arr.indexOf(1)); // 3 // flat() - flatten nested arrays const nested = [1, [2, [3, 4]]]; console.log(nested.flat(2)); // [1, 2, 3, 4]

📦 Object Methods

Methods for working with objects.

Object.keys()

Returns array of object's keys

const user = { name: "John", age: 30 }; console.log(Object.keys(user)); // ["name", "age"]

Object.values()

Returns array of object's values

console.log(Object.values(user)); // ["John", 30]

Object.entries()

Returns array of [key, value] pairs

console.log(Object.entries(user)); // [["name", "John"], ["age", 30]]

Object.assign()

Copy properties to target object

const target = { a: 1 }; const source = { b: 2 }; Object.assign(target, source); // { a: 1, b: 2 }

Object.freeze()

Make object immutable

const obj = { x: 1 }; Object.freeze(obj); obj.x = 2; // No effect console.log(obj.x); // 1

Object.seal()

Prevent adding/removing properties

const obj = { x: 1 }; Object.seal(obj); obj.x = 2; // OK obj.y = 3; // No effect

hasOwnProperty() & in operator

const obj = { name: "John", age: 30 }; // Check if property exists console.log(obj.hasOwnProperty("name")); // true console.log("name" in obj); // true

🔄 JSON Methods

Convert between JavaScript objects and JSON strings.

JSON.stringify()

Convert object to JSON string

const user = { name: "John", age: 30, active: true }; const json = JSON.stringify(user); console.log(json); // '{"name":"John","age":30,"active":true}' // With indentation (pretty print) const pretty = JSON.stringify(user, null, 2); console.log(pretty);

JSON.parse()

Convert JSON string to object

const jsonStr = '{"name":"John","age":30}'; const obj = JSON.parse(jsonStr); console.log(obj.name); // "John" console.log(obj.age); // 30

Real-World Usage

// Store in localStorage const userData = { id: 1, name: "John" }; localStorage.setItem("user", JSON.stringify(userData)); // Retrieve from localStorage const stored = JSON.parse(localStorage.getItem("user")); console.log(stored.name); // Send to API fetch("/api/users", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(userData) });

🌐 Global Methods

Global JavaScript functions available everywhere.

parseInt()

Parse string to integer

console.log(parseInt("123")); // 123 console.log(parseInt("123.45")); // 123 console.log(parseInt("FF", 16)); // 255 (hex)

parseFloat()

Parse string to float

console.log(parseFloat("123.45")); // 123.45 console.log(parseFloat("3.14px")); // 3.14

isNaN()

Check if value is NaN

console.log(isNaN("hello")); // true console.log(isNaN(123)); // false

isFinite()

Check if value is finite number

console.log(isFinite(123)); // true console.log(isFinite(Infinity)); // false

encodeURIComponent()

Encode string for URL

const str = "Hello World!"; console.log(encodeURIComponent(str)); // "Hello%20World%21"

decodeURIComponent()

Decode URL-encoded string

const encoded = "Hello%20World%21"; console.log(decodeURIComponent(encoded)); // "Hello World!"

Timers

// setTimeout() - execute after delay const timeoutId = setTimeout(() => { console.log("Executed after 2 seconds"); }, 2000); // clearTimeout() - cancel timeout clearTimeout(timeoutId); // setInterval() - execute repeatedly const intervalId = setInterval(() => { console.log("Executed every second"); }, 1000); // clearInterval() - cancel interval clearInterval(intervalId);

💡 Pro Tips:

  • ✅ Use map() for transforming arrays
  • ✅ Use filter() for selecting elements
  • ✅ Use reduce() for aggregating values
  • ✅ Use find() instead of filter()[0]
  • ✅ Chain methods for cleaner code
  • ⚠️ Be aware which methods mutate the original (push, pop, splice, sort, reverse)
  • ⚠️ Use slice() or spread operator [...arr] to avoid mutation
