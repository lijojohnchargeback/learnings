Lists, Tuples, Sets, Dictionaries and the Collections Module

📋 Lists

Ordered, mutable sequences. The most versatile data structure in Python.

# Creating lists fruits = ["apple", "banana", "cherry"] numbers = list(range(1, 6)) # [1, 2, 3, 4, 5] mixed = [1, "hello", 3.14, True] empty = [] # Accessing fruits[0] # "apple" fruits[-1] # "cherry" (last item) fruits[1:3] # ["banana", "cherry"] # Modifying fruits[0] = "apricot" fruits.append("date") fruits.insert(1, "blueberry") fruits.extend(["elderberry", "fig"])

📦 Tuples

Ordered, immutable sequences. Great for fixed data and dictionary keys.

# Creating tuples point = (10, 20) single = (42,) # Note the comma! coords = 10, 20, 30 # Parentheses optional # Accessing (same as lists) point[0] # 10 point[-1] # 20 # Unpacking x, y = point # Named tuples (from collections) from collections import namedtuple User = namedtuple("User", ["id", "name", "email"]) user = User(1, "Alice", "alice@example.com") print(user.name) # "Alice"
💡 When to use tuples: Function return values, database records, coordinates, dictionary keys, and anywhere you need immutability.

🔵 Sets

Unordered collections of unique elements. Fast membership testing.

# Creating sets tags = {"python", "backend", "api"} unique = set([1, 2, 2, 3, 3]) # {1, 2, 3} # Operations tags.add("flask") tags.remove("api") # KeyError if missing tags.discard("missing") # No error if missing # Set math a = {1, 2, 3, 4} b = {3, 4, 5, 6} a | b # Union: {1, 2, 3, 4, 5, 6} a & b # Intersection: {3, 4} a - b # Difference: {1, 2} a ^ b # Symmetric diff: {1, 2, 5, 6} # Membership (O(1) average) "python" in tags # True

🗂️ Dictionaries

Key-value mappings. Python's primary data structure for JSON-like data.

# Creating dicts user = {"name": "Alice", "age": 30, "role": "admin"} user = dict(name="Bob", age=25) empty = {} # Accessing user["name"] # "Alice" user.get("email", "N/A") # "N/A" (safe, no KeyError) # Modifying user["email"] = "alice@example.com" user.update({"role": "superadmin", "active": True}) del user["age"] # Iterating for key in user: print(key, user[key]) for key, value in user.items(): print(f"{key}: {value}")

📚 Collections Module

from collections import defaultdict, Counter, deque, OrderedDict # defaultdict — auto-create missing keys word_count = defaultdict(int) for word in ["apple", "banana", "apple"]: word_count[word] += 1 # {"apple": 2, "banana": 1} # Counter — count hashable objects Counter("abracadabra") # Counter({'a': 5, 'b': 2, ...}) counts = Counter(["a", "b", "a", "c", "a"]) counts.most_common(2) # [('a', 3), ('b', 1)] # deque — double-ended queue (O(1) append/pop both ends) queue = deque([1, 2, 3]) queue.appendleft(0) # deque([0, 1, 2, 3]) queue.pop() # 3

📋 Copying & Mutability

import copy # Shallow copy — new container, same inner objects original = [[1, 2], [3, 4]] shallow = original.copy() # or list(original) shallow[0].append(99) # original is also changed: [[1, 2, 99], [3, 4]] # Deep copy — fully independent deep = copy.deepcopy(original) deep[0].append(100) # original unchanged
⚠️ Common Bug: Assigning b = a creates a reference, not a copy. Use .copy() or copy.deepcopy() for nested structures.
