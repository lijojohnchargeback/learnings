📖 Classes & Objects

class User: species = "Homo sapiens" # Class attribute (shared) def __init__(self, name, email): self.name = name # Instance attribute self.email = email def greet(self): return f"Hello, I'm {self.name}" # Creating objects user = User("Alice", "alice@example.com") print(user.greet()) # "Hello, I'm Alice" print(User.species) # "Homo sapiens"

🔧 __init__ & Attributes

class Product: def __init__(self, name: str, price: float, stock: int = 0): self.name = name self.price = price self._stock = stock # Protected (convention) self.__sku = "AUTO-001" # Name-mangled (private) def __repr__(self): return f"Product(name={self.name!r}, price={self.price})" def __str__(self): return f"{self.name} - ${self.price:.2f}" product = Product("Laptop", 999.99, stock=10) print(repr(product)) # Product(name='Laptop', price=999.99) print(str(product)) # Laptop - $999.99

⚙️ Instance, Class & Static Methods

class DatabaseConnection: _pool_size = 5 def __init__(self, url): self.url = url # Instance method — operates on self def connect(self): return f"Connected to {self.url}" # Class method — operates on class, receives cls @classmethod def from_env(cls): import os return cls(os.environ["DATABASE_URL"]) @classmethod def set_pool_size(cls, size): cls._pool_size = size # Static method — no access to self or cls @staticmethod def validate_url(url): return url.startswith(("postgres://", "mysql://")) db = DatabaseConnection.from_env()

🏷️ Properties

class Temperature: def __init__(self, celsius): self._celsius = celsius @property def celsius(self): return self._celsius @celsius.setter def celsius(self, value): if value < -273.15: raise ValueError("Below absolute zero") self._celsius = value @property def fahrenheit(self): return self._celsius * 9/5 + 32 temp = Temperature(25) print(temp.fahrenheit) # 77.0 temp.celsius = 30 # Uses setter

🧬 Inheritance

class Animal: def __init__(self, name): self.name = name def speak(self): raise NotImplementedError class Dog(Animal): def speak(self): return f"{self.name} says Woof!" class Cat(Animal): def speak(self): return f"{self.name} says Meow!" # Multiple inheritance class FlyingMixin: def fly(self): return "Flying high!" class Duck(Animal, FlyingMixin): def speak(self): return f"{self.name} says Quack!" # Method Resolution Order (MRO) print(Duck.__mro__)

🔄 Polymorphism

def animal_sound(animal: Animal): print(animal.speak()) animals = [Dog("Rex"), Cat("Whiskers"), Duck("Donald")] for animal in animals: animal_sound(animal) # Rex says Woof! # Whiskers says Meow! # Donald says Quack!

📐 Abstract Base Classes

from abc import ABC, abstractmethod class Repository(ABC): @abstractmethod def find_by_id(self, id: int): pass @abstractmethod def save(self, entity): pass class UserRepository(Repository): def find_by_id(self, id: int): return db.query(User).get(id) def save(self, entity): db.session.add(entity) db.session.commit() # Cannot instantiate Repository directly # repo = Repository() # TypeError!

✨ Dataclasses (Python 3.7+)

from dataclasses import dataclass, field from typing import List @dataclass class UserDTO: id: int name: str email: str roles: List[str] = field(default_factory=list) is_active: bool = True user = UserDTO(1, "Alice", "alice@example.com") user.roles.append("admin") # Frozen (immutable) @dataclass(frozen=True) class Config: host: str port: int = 8000
💡 Backend tip: Use dataclasses or Pydantic models for request/response DTOs in APIs.
