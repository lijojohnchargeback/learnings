📝 Python Type Hints & Static Typing

Annotations, generics, protocols, Pydantic, and checkers for safer backends

🎯 Why Typing?

Type hints are optional at runtime — CPython ignores them for execution. Tools like mypy, pyright, and IDEs use them to catch bugs early and document APIs.

BenefitExample
Catch bugs before prodPassing None where str expected
Better editor helpAutocomplete on return types
Living docsFunction signatures explain contracts
Safer refactorsRename / change shapes with confidence

✍️ Basic Annotations

def greet(name: str, times: int = 1) -> str: return ("Hello, " + name + "! ") * times age: int = 30 pi: float = 3.14 active: bool = True data: bytes = b"raw" # Python 3.9+: builtins are generic def first(items: list[str]) -> str: return items[0] # Python 3.8 and earlier: from typing import List def first_old(items: List[str]) -> str: return items[0]

📚 Collections

from collections.abc import Mapping, Sequence, Iterable, Callable def total(nums: Sequence[int]) -> int: return sum(nums) def merge(a: Mapping[str, int], b: Mapping[str, int]) -> dict[str, int]: return {**a, **b} Handler = Callable[[str, int], bool] def apply(fn: Handler, name: str, n: int) -> bool: return fn(name, n) # tuple length/types Pair = tuple[str, int] point: tuple[float, float] = (1.0, 2.0) row: tuple[str, ...] = ("a", "b", "c") # variable length

🔀 Optional, Union & Literals

# Python 3.10+ union with | def find_user(user_id: int) -> dict | None: ... def parse(value: str | int) -> int: return int(value) from typing import Literal, Final Mode = Literal["r", "w", "a"] def open_file(path: str, mode: Mode) -> None: ... MAX_RETRIES: Final = 3 # Type narrowing def length(x: str | None) -> int: if x is None: return 0 return len(x) # x is str here

🧬 Generics

from typing import TypeVar, Generic T = TypeVar("T") class Repo(Generic[T]): def __init__(self) -> None: self._items: list[T] = [] def add(self, item: T) -> None: self._items.append(item) def get(self, index: int) -> T: return self._items[index] users: Repo[dict] = Repo() # Python 3.12+ shorter syntax class Stack[T]: def __init__(self) -> None: self._data: list[T] = [] def push(self, item: T) -> None: self._data.append(item) def pop(self) -> T: return self._data.pop() # Bounded TypeVar from typing import TypeVar Number = TypeVar("Number", int, float) def double(x: Number) -> Number: return x * 2

🦆 Protocols

from typing import Protocol class SupportsClose(Protocol): def close(self) -> None: ... def cleanup(resource: SupportsClose) -> None: resource.close() # Structural — no inheritance required class Connection: def close(self) -> None: ... cleanup(Connection())

📋 TypedDict, NamedTuple, NewType

from typing import TypedDict, NamedTuple, NewType, NotRequired class UserDict(TypedDict): id: int email: str nickname: NotRequired[str] user: UserDict = {"id": 1, "email": "a@b.com"} class Point(NamedTuple): x: float y: float UserId = NewType("UserId", int) def load_user(user_id: UserId) -> None: ... load_user(UserId(42)) # load_user(42) # type checker error

✅ Pydantic (Runtime Validation)

Type hints document intent; Pydantic enforces them at runtime — ideal for API request/response models.

from pydantic import BaseModel, EmailStr, Field, field_validator class CreateUser(BaseModel): email: EmailStr age: int = Field(ge=0, le=150) roles: list[str] = Field(default_factory=list) @field_validator("roles") @classmethod def non_empty_roles(cls, v: list[str]) -> list[str]: if not v: return ["user"] return v payload = CreateUser(email="dev@example.com", age=25) print(payload.model_dump()) # FastAPI uses these models for parsing + OpenAPI docs

🔍 mypy & pyright

# pyproject.toml (example) # [tool.mypy] # python_version = "3.12" # strict = true # warn_return_any = true # exclude = ["migrations/"] # CLI # mypy src/ # pyright src/ # Escape hatches (use sparingly) from typing import Any, cast, TYPE_CHECKING def parse_raw(data: Any) -> dict[str, str]: return cast(dict[str, str], data) if TYPE_CHECKING: from expensive import HeavyModel # import only for type checkers

📌 Best Practices

  • Annotate public APIs first (services, routers, domain models)
  • Prefer list[str] / X | None (3.10+) over legacy typing aliases
  • Use collections.abc for parameters (Sequence, Mapping)
  • Keep Any rare — it turns the checker off
  • Pair hints with Pydantic/FastAPI for HTTP boundaries
  • Enable strict mode gradually; don’t block the team overnight
💡 Remember: Hints don’t replace tests. They catch a different class of mistakes — shape and contract errors — before runtime.
