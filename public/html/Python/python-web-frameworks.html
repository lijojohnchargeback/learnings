Flask, FastAPI, and Django for backend development

📊 Framework Comparison

FrameworkTypeBest ForLearning Curve
Flask Microframework Small-to-medium APIs, prototypes, microservices Easy
FastAPI Modern ASGI framework High-performance APIs, async, auto-docs Easy-Medium
Django Full-stack framework Large apps, admin panels, content sites Medium-Hard
💡 Recommendation: Use FastAPI for new APIs, Flask for simple services, and Django when you need batteries-included (ORM, admin, auth).

🔄 WSGI vs ASGI

ProtocolDescriptionFrameworks
WSGI Web Server Gateway Interface — synchronous, one request per worker Flask, Django (default)
ASGI Asynchronous Server Gateway Interface — async, WebSockets, long-lived connections FastAPI, Starlette, Django Channels
# WSGI servers: Gunicorn, uWSGI gunicorn app:app --workers 4 --bind 0.0.0.0:8000 # ASGI servers: Uvicorn, Hypercorn uvicorn app:app --host 0.0.0.0 --port 8000 --workers 4

🧪 Flask

# app.py from flask import Flask, request, jsonify app = Flask(__name__) @app.route("/") def home(): return jsonify({"message": "Hello, Flask!"}) @app.route("/users/<int:user_id>", methods=["GET"]) def get_user(user_id): return jsonify({"id": user_id, "name": "Alice"}) @app.route("/users", methods=["POST"]) def create_user(): data = request.get_json() return jsonify({"id": 1, **data}), 201 @app.errorhandler(404) def not_found(e): return jsonify({"error": "Not found"}), 404 if __name__ == "__main__": app.run(debug=True, port=5000)
# Install & run pip install flask python app.py # or flask --app app run --debug

📁 Flask Blueprints

# routes/users.py from flask import Blueprint, jsonify users_bp = Blueprint("users", __name__, url_prefix="/api/users") @users_bp.route("/") def list_users(): return jsonify([{"id": 1, "name": "Alice"}]) @users_bp.route("/<int:id>") def get_user(id): return jsonify({"id": id, "name": "Alice"}) # app.py from flask import Flask from routes.users import users_bp app = Flask(__name__) app.register_blueprint(users_bp)

⚡ FastAPI

# main.py from fastapi import FastAPI, HTTPException, Depends from pydantic import BaseModel app = FastAPI(title="My API", version="1.0.0") class UserCreate(BaseModel): name: str email: str age: int | None = None class UserResponse(BaseModel): id: int name: str email: str @app.get("/") async def root(): return {"message": "Hello, FastAPI!"} @app.get("/users/{user_id}", response_model=UserResponse) async def get_user(user_id: int): if user_id != 1: raise HTTPException(status_code=404, detail="User not found") return UserResponse(id=1, name="Alice", email="alice@example.com") @app.post("/users", response_model=UserResponse, status_code=201) async def create_user(user: UserCreate): return UserResponse(id=1, name=user.name, email=user.email)
# Install & run pip install fastapi uvicorn uvicorn main:app --reload # Auto-generated docs at http://localhost:8000/docs

✅ FastAPI Validation & Dependency Injection

from fastapi import Depends, Query, Header from pydantic import BaseModel, EmailStr, Field class UserCreate(BaseModel): name: str = Field(..., min_length=2, max_length=50) email: EmailStr age: int = Field(ge=0, le=150) # Dependency injection def get_db(): db = SessionLocal() try: yield db finally: db.close() @app.get("/items") async def list_items( skip: int = Query(0, ge=0), limit: int = Query(10, le=100), db = Depends(get_db) ): return db.query(Item).offset(skip).limit(limit).all()
💡 Auto-docs: FastAPI generates interactive Swagger UI at /docs and ReDoc at /redoc automatically.

🎸 Django

# Project setup pip install django django-admin startproject myproject cd myproject python manage.py startapp users # settings.py — add 'users' to INSTALLED_APPS # users/models.py from django.db import models class User(models.Model): name = models.CharField(max_length=100) email = models.EmailField(unique=True) created_at = models.DateTimeField(auto_now_add=True) def __str__(self): return self.name # users/views.py from django.http import JsonResponse from .models import User def user_list(request): users = list(User.objects.values("id", "name", "email")) return JsonResponse({"users": users}) # users/urls.py from django.urls import path from . import views urlpatterns = [ path("users/", views.user_list), ]
